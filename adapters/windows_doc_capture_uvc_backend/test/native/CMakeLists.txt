cmake_minimum_required(VERSION 3.14)
project(doc_capture_sampler_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(sampler_test sampler_test.cpp)
if(MSVC)
  target_compile_options(sampler_test PRIVATE /W4 /WX)

endif()

add_test(NAME sampler_bounds_pixels COMMAND sampler_test)
